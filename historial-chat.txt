```typescript
import { Pool, PoolConfig } from 'pg';

const poolConfig: PoolConfig = {
  host: process.env.DB_HOST || 'localhost',
  port: parseInt(process.env.DB_PORT || '5432', 10),
  user: process.env.DB_USER || 'postgres',
  password: process.env.DB_PASSWORD || 'postgres',
  database: process.env.DB_NAME || 'clientes_db',
  max: 20,
  idleTimeoutMillis: 30000,
  connectionTimeoutMillis: 2000,
};

const pool = new Pool(poolConfig);

pool.on('error', (err) => {
  console.error('Unexpected error on idle client', err);
  process.exit(-1);
});

export const query = async <T>(
  text: string,
  params?: unknown[]
): Promise<T[]> => {
  const client = await pool.connect();
  try {
    const result = await client.query(text, params);
    return result.rows as T[];
  } finally {
    client.release();
  }
};

export const queryOne = async <T>(
  text: string,
  params?: unknown[]
): Promise<T | null> => {
  const client = await pool.connect();
  try {
    const result = await client.query(text, params);
    return (result.rows[0] as T) || null;
  } finally {
    client.release();
  }
};

export const execute = async (
  text: string,
  params?: unknown[]
): Promise<number> => {
  const client = await pool.connect();
  try {
    const result = await client.query(text, params);
    return result.rowCount || 0;
  } finally {
    client.release();
  }
};

export const getClient = () => pool.connect();

export default pool;
```

### backend/src/persistence/cliente.repository.ts
```typescript
import { query, queryOne, execute } from './database';
import { Cliente, CreateClienteDTO } from '../models/cliente.model';

interface ClienteRow {
  codigo_cliente: number;
  nombre_cliente: string;
  direccion_cliente: string;
  telefono_cliente: string;
}

const mapRowToCliente = (row: ClienteRow): Cliente => ({
  codigoCliente: row.codigo_cliente,
  nombreCliente: row.nombre_cliente,
  direccionCliente: row.direccion_cliente,
  telefonoCliente: row.telefono_cliente,
});

export const findAll = async (): Promise<Cliente[]> => {
  const rows = await query<ClienteRow>(
    'SELECT codigo_cliente, nombre_cliente, direccion_cliente, telefono_cliente FROM clientes ORDER BY codigo_cliente'
  );
  return rows.map(mapRowToCliente);
};

export const findByCodigo = async (codigo: number): Promise<Cliente | null> => {
  const row = await queryOne<ClienteRow>(
    'SELECT codigo_cliente, nombre_cliente, direccion_cliente, telefono_cliente FROM clientes WHERE codigo_cliente = $1',
    [codigo]
  );
  return row ? mapRowToCliente(row) : null;
};

export const create = async (cliente: CreateClienteDTO): Promise<Cliente> => {
  const row = await queryOne<ClienteRow>(
    'INSERT INTO clientes (nombre_cliente, direccion_cliente, telefono_cliente) VALUES ($1, $2, $3) RETURNING codigo_cliente, nombre_cliente, direccion_cliente, telefono_cliente',
    [cliente.nombreCliente, cliente.direccionCliente, cliente.telefonoCliente]
  );
  if (!row) {
    throw new Error('Error al crear el cliente');
  }
  return mapRowToCliente(row);
};

export const update = async (
  codigo: number,
  cliente: Partial<CreateClienteDTO>
): Promise<Cliente | null> => {
  const fields: string[] = [];
  const values: unknown[] = [];
  let paramIndex = 1;

  if (cliente.nombreCliente !== undefined) {
    fields.push(`nombre_cliente = $${paramIndex++}`);
    values.push(cliente.nombreCliente);
  }
  if (cliente.direccionCliente !== undefined) {
    fields.push(`direccion_cliente = $${paramIndex++}`);
    values.push(cliente.direccionCliente);
  }
  if (cliente.telefonoCliente !== undefined) {
    fields.push(`telefono_cliente = $${paramIndex++}`);
    values.push(cliente.telefonoCliente);
  }

  if (fields.length === 0) {
    return findByCodigo(codigo);
  }

  values.push(codigo);
  const row = await queryOne<ClienteRow>(
    `UPDATE clientes SET ${fields.join(', ')} WHERE codigo_cliente = $${paramIndex} RETURNING codigo_cliente, nombre_cliente, direccion_cliente, telefono_cliente`,
    values
  );
  return row ? mapRowToCliente(row) : null;
};

export const remove = async (codigo: number): Promise<boolean> => {
  const rowCount = await execute(
    'DELETE FROM clientes WHERE codigo_cliente = $1',
    [codigo]
  );
  return rowCount > 0;
};

export const existsByCodigo = async (codigo: number): Promise<boolean> => {
  const row = await queryOne<{ count: number }>(
    'SELECT COUNT(*) as count FROM clientes WHERE codigo_cliente = $1',
    [codigo]
  );
  return row ? parseInt(String(row.count), 10) > 0 : false;
};
```

### backend/src/database/init.ts

```typescript
import { Pool } from 'pg';
import * as fs from 'fs';
import * as path from 'path';

const pool = new Pool({
  host: process.env.DB_HOST || 'localhost',
  port: parseInt(process.env.DB_PORT || '5432', 10),
  user: process.env.DB_USER || 'postgres',
  password: process.env.DB_PASSWORD || 'postgres',
  database: process.env.DB_NAME || 'clientes_db',
});

const createTableQuery = `
  CREATE TABLE IF NOT EXISTS clientes (
    codigo_cliente SERIAL PRIMARY KEY,
    nombre_cliente VARCHAR(100) NOT NULL,
    direccion_cliente VARCHAR(200) NOT NULL,
    telefono_cliente VARCHAR(20) NOT NULL
  );
`;

const seedData = async (): Promise<void> => {
  const seedPath = path.join(__dirname, '../../data/clientes.json');
  if (!fs.existsSync(seedPath)) {
    console.log('No seed data file found');
    return;
  }

  const rawData = fs.readFileSync(seedPath, 'utf-8');
  const clientes = JSON.parse(rawData) as Array<{
    codigo_cliente: number;
    nombre_cliente: string;
    direccion_cliente: string;
    telefono_cliente: string;
  }>;

  const client = await pool.connect();
  try {
    for (const cliente of clientes) {
      await client.query(
        `INSERT INTO clientes (codigo_cliente, nombre_cliente, direccion_cliente, telefono_cliente)
         VALUES ($1, $2, $3, $4)
         ON CONFLICT (codigo_cliente) DO NOTHING`,
        [
          cliente.codigo_cliente,
          cliente.nombre_cliente,
          cliente.direccion_cliente,
          cliente.telefono_cliente,
        ]
      );
    }
    console.log(`Seeded ${clientes.length} clients`);
  } finally {
    client.release();
  }
};

const initDatabase = async (): Promise<void> => {
  try {
    console.log('Initializing database...');
    await pool.query(createTableQuery);
    console.log('Table created successfully');

    console.log('Seeding data...');
    await seedData();

    console.log('Database initialization completed');
  } catch (error) {
    console.error('Error initializing database:', error);
    throw error;
  } finally {
    await pool.end();
  }
};

void initDatabase();
```

```typescript
type ValidationErrors = string[];

type RouteParams = { codigo: string };

const parseCodigo = (raw: string | string[]): number => {
  const value = Array.isArray(raw) ? raw[0] : raw;
  return parseInt(value, 10);

const validateCliente = (data: ClienteDTO, isUpdate = false): ValidationErrors => {
  const errors: ValidationErrors = [];

  if (!isUpdate || data.nombreCliente !== undefined) {
    if (!data.nombreCliente || data.nombreCliente.trim().length === 0) {
      errors.push('El nombre del cliente es requerido');
    } else if (data.nombreCliente.length > 100) {
      errors.push('El nombre del cliente no puede exceder 100 caracteres');
    }
  }

  if (!isUpdate || data.direccionCliente !== undefined) {
    if (!data.direccionCliente || data.direccionCliente.trim().length === 0) {
      errors.push('La dirección del cliente es requerida');
    } else if (data.direccionCliente.length > 200) {
      errors.push('La dirección del cliente no puede exceder 200 caracteres');
    }
  }

  if (!isUpdate || data.telefonoCliente !== undefined) {
    if (!data.telefonoCliente || data.telefonoCliente.trim().length === 0) {
      errors.push('El teléfono del cliente es requerido');
    } else if (data.telefonoCliente.length < 7 || data.telefonoCliente.length > 20) {
      errors.push('El teléfono debe tener entre 7 y 20 caracteres');
    } else if (!/^[0-9+\-\s()]+$/.test(data.telefonoCliente)) {
      errors.push('El teléfono solo puede contener números, espacios, guiones, paréntesis y el signo +');
    }
  }

  return errors;
};

export const getAllClientes = async (_req: Request, res: Response): Promise<void> => {
  try {
    const clientes = await clienteRepository.findAll();
    const response: ApiResponse<Cliente[]> = {
      success: true,
      data: clientes,
    };
    res.json(response);
  } catch (error) {
    const message = error instanceof Error ? error.message : 'Error al obtener los clientes';
    const response: ApiResponse<never> = {
      success: false,
      error: message,
    };
    res.status(500).json(response);
  }
};

export const getClienteByCodigo = async (req: Request, res: Response): Promise<void> => {

    res.status(500).json(response);

export const createCliente = async (req: Request, res: Response): Promise<void> => {
  try {
    const { nombreCliente, direccionCliente, telefonoCliente } = req.body;
    const clienteData: ClienteDTO = { nombreCliente, direccionCliente, telefonoCliente };

    const validationErrors = validateCliente(clienteData);
    if (validationErrors.length > 0) {
      const response: ApiResponse<never> = {
        success: false,
        error: validationErrors.join(', '),
      };
      res.status(400).json(response);
      return;
    }

    const newCliente: CreateClienteDTO = {
      nombreCliente: nombreCliente.trim(),
      direccionCliente: direccionCliente.trim(),
      telefonoCliente: telefonoCliente.trim(),
    };

    const cliente = await clienteRepository.create(newCliente);
    const response: ApiResponse<Cliente> = {
      success: true,
      data: cliente,
      message: 'Cliente creado exitosamente',
    };
    res.status(201).json(response);
  } catch (error) {
    const message = error instanceof Error ? error.message : 'Error al crear el cliente';
    const response: ApiResponse<never> = {
      success: false,
      error: message,
    };
    res.status(500).json(response);
  }
};

export const updateCliente = async (req: Request, res: Response): Promise<void> => {
  try {
    const codigo = parseCodigo(req.params.codigo);
    if (isNaN(codigo)) {
      const response: ApiResponse<never> = {
        success: false,
        error: 'El código del cliente debe ser un número válido',
      };
      res.status(400).json(response);
      return;
    }

    const existingCliente = await clienteRepository.findByCodigo(codigo);
    if (!existingCliente) {
      const response: ApiResponse<never> = {
        success: false,
        error: `No se encontró el cliente con código ${codigo}`,
      };
      res.status(404).json(response);
      return;
    }

    const { nombreCliente, direccionCliente, telefonoCliente } = req.body;
    const clienteData: ClienteDTO = { nombreCliente, direccionCliente, telefonoCliente };

    const validationErrors = validateCliente(clienteData, true);
    if (validationErrors.length > 0) {
      const response: ApiResponse<never> = {
        success: false,
        error: validationErrors.join(', '),
      };
      res.status(400).json(response);
      return;
    }

    const updateData: Partial<CreateClienteDTO> = {};
    if (nombreCliente !== undefined) {
      updateData.nombreCliente = nombreCliente.trim();
    }
    if (direccionCliente !== undefined) {
      updateData.direccionCliente = direccionCliente.trim();
    }
    if (telefonoCliente !== undefined) {
      updateData.telefonoCliente = telefonoCliente.trim();
    }

    const cliente = await clienteRepository.update(codigo, updateData);
    if (!cliente) {
      const response: ApiResponse<never> = {
        success: false,
        error: 'Error al actualizar el cliente',
      };
      res.status(500).json(response);
      return;
    }

    const response: ApiResponse<Cliente> = {
      success: true,
      data: cliente,
      message: 'Cliente actualizado exitosamente',
    };
    res.json(response);
  } catch (error) {
    const message = error instanceof Error ? error.message : 'Error al actualizar el cliente';
    const response: ApiResponse<never> = {
      success: false,
      error: message,
    };
    res.status(500).json(response);
  }
};
```

### frontend/src/services/cliente.service.ts
```typescript
import { Injectable, signal } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable, tap } from 'rxjs';
import { Cliente, ClienteDTO, ApiResponse } from '../models/cliente.model';

@Injectable({
  providedIn: 'root',
})
export class ClienteService {
  private readonly apiUrl = 'http://localhost:3000/clientes';

  readonly loading = signal<boolean>(false);
  readonly error = signal<string | null>(null);
  readonly success = signal<string | null>(null);

  constructor(private http: HttpClient) {}

  getAll(): Observable<ApiResponse<Cliente[]>> {
    this.loading.set(true);
    this.error.set(null);
    return this.http.get<ApiResponse<Cliente[]>>(this.apiUrl).pipe(
      tap({
        next: () => this.loading.set(false),
        error: () => this.loading.set(false),
      })
    );
  }

  getByCodigo(codigo: number): Observable<ApiResponse<Cliente>> {
    this.loading.set(true);
    this.error.set(null);
    return this.http.get<ApiResponse<Cliente>>(`${this.apiUrl}/${codigo}`).pipe(
      tap({
        next: () => this.loading.set(false),
        error: () => this.loading.set(false),
      })
    );
  }

  create(cliente: ClienteDTO): Observable<ApiResponse<Cliente>> {
    this.loading.set(true);
    this.error.set(null);
    this.success.set(null);
    return this.http.post<ApiResponse<Cliente>>(this.apiUrl, cliente).pipe(
      tap({
        next: (response) => {
          this.loading.set(false);
          if (response.success && response.message) {
            this.success.set(response.message);
          }
        },
        error: () => this.loading.set(false),
      })
    );
  }

  update(codigo: number, cliente: ClienteDTO): Observable<ApiResponse<Cliente>> {
    this.loading.set(true);
    this.error.set(null);
    this.success.set(null);
    return this.http.put<ApiResponse<Cliente>>(`${this.apiUrl}/${codigo}`, cliente).pipe(
      tap({
        next: (response) => {
          this.loading.set(false);
          if (response.success && response.message) {
            this.success.set(response.message);
          }
        },
        error: () => this.loading.set(false),
      })
    );
  }

  delete(codigo: number): Observable<ApiResponse<null>> {
    this.loading.set(true);
    this.error.set(null);
    this.success.set(null);
    return this.http.delete<ApiResponse<null>>(`${this.apiUrl}/${codigo}`).pipe(
      tap({
        next: (response) => {
          this.loading.set(false);
          if (response.success && response.message) {
            this.success.set(response.message);
          }
        },
        error: () => this.loading.set(false),
      })
    );
  }

  clearMessages(): void {
    this.error.set(null);
    this.success.set(null);
  }

  setError(message: string): void {
    this.error.set(message);
    this.loading.set(false);
  }
}
```

```typescript
import { Component, EventEmitter, Input, OnChanges, Output, SimpleChanges, signal } from '@angular/core';
import { CommonModule } from '@angular/common';
import { ReactiveFormsModule, FormBuilder, FormGroup, Validators } from '@angular/forms';
import { ClienteService } from '../../services/cliente.service';
import { Cliente } from '../../models/cliente.model';

@Component({
  selector: 'app-cliente-form',
  standalone: true,
  imports: [CommonModule, ReactiveFormsModule],
  templateUrl: './cliente-form.component.html',
  styleUrl: './cliente-form.component.css',
})
export class ClienteFormComponent implements OnChanges {
  @Input() cliente: Cliente | null = null;
  @Output() clienteGuardado = new EventEmitter<void>();
  @Output() cancelarEdicion = new EventEmitter<void>();

  clienteForm: FormGroup;
  isSubmitting = signal<boolean>(false);

  constructor(
    private fb: FormBuilder,
    public clienteService: ClienteService
  ) {
    this.clienteForm = this.createForm();
  }

  ngOnChanges(changes: SimpleChanges): void {
    if (changes['cliente'] && this.cliente) {
      this.populateForm(this.cliente);
    } else if (changes['cliente'] && !this.cliente) {
      this.resetForm();
    }
  }

  private createForm(): FormGroup {
    return this.fb.group({
      nombreCliente: ['', [Validators.required, Validators.maxLength(100)]],
      direccionCliente: ['', [Validators.required, Validators.maxLength(200)]],
      telefonoCliente: [
        '',
        [
          Validators.required,
          Validators.minLength(7),
          Validators.maxLength(20),
          Validators.pattern(/^[0-9+\-\s()]+$/),
        ],
      ],
    });
  }

  private populateForm(cliente: Cliente): void {
    this.clienteForm.patchValue({
      nombreCliente: cliente.nombreCliente,
      direccionCliente: cliente.direccionCliente,
      telefonoCliente: cliente.telefonoCliente,
    });
  }

  private resetForm(): void {
    this.clienteForm.reset();
    this.clienteService.clearMessages();
  }

  get isEditing(): boolean {
    return this.cliente !== null;
  }

  get formTitle(): string {
    return this.isEditing ? 'Editar Cliente' : 'Nuevo Cliente';
  }

  get submitButtonText(): string {
    return this.isEditing ? 'Actualizar' : 'Crear';
  }

  onSubmit(): void {
    if (this.clienteForm.invalid) {
      this.clienteForm.markAllAsTouched();
      return;
    }

    this.isSubmitting.set(true);
    const formData = this.clienteForm.value as Cliente;

    const operation$ = this.isEditing && this.cliente
      ? this.clienteService.update(this.cliente.codigoCliente, formData)
      : this.clienteService.create(formData);

    operation$.subscribe({
      next: (response) => {
        this.isSubmitting.set(false);
        if (response.success) {
          this.resetForm();
          this.clienteGuardado.emit();
        } else {
          this.clienteService.setError(response.error || 'Error al guardar el cliente');
        }
      },
      error: (err) => {
        this.isSubmitting.set(false);
        const message = err.error?.error || 'Error de conexión con el servidor';
        this.clienteService.setError(message);
      },
    });
  }

  onCancel(): void {
    this.resetForm();
    this.cancelarEdicion.emit();
  }

  isFieldInvalid(fieldName: string): boolean {
    const field = this.clienteForm.get(fieldName);
    return field !== null && field.invalid && field.touched;
  }

  getFieldError(fieldName: string): string {
    const field = this.clienteForm.get(fieldName);
    if (!field || !field.errors || !field.touched) return '';

    if (field.errors['required']) {
      return `${this.getFieldLabel(fieldName)} es requerido`;
    }
    if (field.errors['maxlength']) {
      return `${this.getFieldLabel(fieldName)} no puede exceder ${field.errors['maxlength'].requiredLength} caracteres`;
    }
    if (field.errors['minlength']) {
      return `${this.getFieldLabel(fieldName)} debe tener al menos ${field.errors['minlength'].requiredLength} caracteres`;
    }
    if (field.errors['pattern']) {
      return `${this.getFieldLabel(fieldName)} solo puede contener números, espacios, guiones, paréntesis y el signo +`;
    }
    return '';
  }

  private getFieldLabel(fieldName: string): string {
    const labels: Record<string, string> = {
      nombreCliente: 'El nombre',
      direccionCliente: 'La dirección',
      telefonoCliente: 'El teléfono',
    };
    return labels[fieldName] || fieldName;
  }
}
```

```typescript
import { Component, EventEmitter, OnInit, Output, signal } from '@angular/core';
import { CommonModule } from '@angular/common';
import { ClienteService } from '../../services/cliente.service';
import { Cliente } from '../../models/cliente.model';

@Component({
  selector: 'app-cliente-list',
  standalone: true,
  imports: [CommonModule],
  templateUrl: './cliente-list.component.html',
  styleUrl: './cliente-list.component.css',
})
export class ClienteListComponent implements OnInit {
  @Output() editarCliente = new EventEmitter<Cliente>();

  clientes = signal<Cliente[]>([]);
  confirmDelete = signal<Cliente | null>(null);

  constructor(public clienteService: ClienteService) {}

  ngOnInit(): void {
    this.loadClientes();
  }

  loadClientes(): void {
    this.clienteService.getAll().subscribe({
      next: (response) => {
        if (response.success && response.data) {
          this.clientes.set(response.data);
        } else {
          this.clienteService.setError(response.error || 'Error al cargar los clientes');
        }
      },
      error: (err) => {
        const message = err.error?.error || 'Error de conexión con el servidor';
        this.clienteService.setError(message);
      },
    });
  }

  onEdit(cliente: Cliente): void {
    this.editarCliente.emit(cliente);
  }

  requestDelete(cliente: Cliente): void {
    this.confirmDelete.set(cliente);
  }

  confirmDeleteAction(): void {
    const cliente = this.confirmDelete();
    if (!cliente) return;

    this.clienteService.delete(cliente.codigoCliente).subscribe({
      next: (response) => {
        if (response.success) {
          this.loadClientes();
        } else {
          this.clienteService.setError(response.error || 'Error al eliminar el cliente');
        }
        this.confirmDelete.set(null);
      },
      error: (err) => {
        const message = err.error?.error || 'Error de conexión con el servidor';
        this.clienteService.setError(message);
        this.confirmDelete.set(null);
      },
    });
  }

  cancelDelete(): void {
    this.confirmDelete.set(null);
  }

  formatTelefono(telefono: string): string {
    return telefono;
  }

  trackByCodigo(_index: number, cliente: Cliente): number {
    return cliente.codigoCliente;
  }
}
```

```sql
-- ============================================================

-- ============================================================
```

```sql
-- Eliminar tabla si existe (orden por dependencias)
DROP TABLE IF EXISTS clientes CASCADE;

-- Crear tabla clientes
CREATE TABLE clientes (
    codigo_cliente SERIAL PRIMARY KEY,
    nombre_cliente VARCHAR(100) NOT NULL,
    direccion_cliente VARCHAR(200) NOT NULL,
    telefono_cliente VARCHAR(20) NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_clientes_nombre ON clientes(nombre_cliente);

COMMENT ON COLUMN clientes.codigo_cliente IS 'Código único identificador del cliente (autoincremental)';
COMMENT ON COLUMN clientes.nombre_cliente IS 'Nombre o razón social del cliente';
COMMENT ON COLUMN clientes.direccion_cliente IS 'Dirección fiscal o de contacto del cliente';
COMMENT ON COLUMN clientes.telefono_cliente IS 'Número de teléfono de contacto del cliente';

\echo 'Tabla clientes creada exitosamente'
```

                  CONTINUACIÓN - ROUND 4: SERIAL autoincrementable
================================================================================

USUARIO:
el codigo de cliente es autoincrementable?

================================================================================

ASISTENTE:
Sí, es autoincrementable. La columna usa `SERIAL` en PostgreSQL:

```sql
codigo_cliente SERIAL PRIMARY KEY
```

El `INSERT` en el repositorio (`cliente.repository.ts:29`) no incluye `codigo_cliente` — PostgreSQL genera el valor automáticamente:

```sql
INSERT INTO clientes (nombre_cliente, direccion_cliente, telefono_cliente) VALUES ($1, $2, $3) RETURNING ...
```

Y el DTO de creación excluye el código:

```typescript
export type CreateClienteDTO = Omit<Cliente, 'codigoCliente'>;
```

El frontend tampoco lo solicita en el formulario.

================================================================================

USUARIO:
al intentar agregar un cliente desde la web indica llave duplicada viola restricción de unicidad «clientes_pkey»

================================================================================

ASISTENTE:
El problema es que la semilla inserta clientes con `codigo_cliente` explícitos (1-5), pero la secuencia `SERIAL` no se actualiza y empieza en 1, causando el conflicto.

Se actualizó `backend/src/database/init.ts` para sincronizar la secuencia después de insertar la semilla:

```typescript
console.log('Resetting sequence...');
await pool.query(
  `SELECT setval('clientes_codigo_cliente_seq', COALESCE((SELECT MAX(codigo_cliente) FROM clientes), 1))`
);
console.log('Sequence reset successfully');
```

```typescript
import { query, queryOne, execute } from './database';
import { Cliente, CreateClienteDTO } from '../models/cliente.model';

interface ClienteRow {
  codigo_cliente: number;
  nombre_cliente: string;
  direccion_cliente: string;
  telefono_cliente: string;
}

const mapRowToCliente = (row: ClienteRow): Cliente => ({
  codigoCliente: row.codigo_cliente,
  nombreCliente: row.nombre_cliente,
  direccionCliente: row.direccion_cliente,
  telefonoCliente: row.telefono_cliente,
});

export const findAll = async (): Promise<Cliente[]> => {
  const rows = await query<ClienteRow>(
    'SELECT codigo_cliente, nombre_cliente, direccion_cliente, telefono_cliente FROM clientes ORDER BY codigo_cliente'
  );
  return rows.map(mapRowToCliente);
};

export const findByCodigo = async (codigo: number): Promise<Cliente | null> => {
  const row = await queryOne<ClienteRow>(
    'SELECT codigo_cliente, nombre_cliente, direccion_cliente, telefono_cliente FROM clientes WHERE codigo_cliente = $1',
    [codigo]
  );
  return row ? mapRowToCliente(row) : null;
};

export const create = async (cliente: CreateClienteDTO): Promise<Cliente> => {
  const row = await queryOne<ClienteRow>(
    'INSERT INTO clientes (nombre_cliente, direccion_cliente, telefono_cliente) VALUES ($1, $2, $3) RETURNING codigo_cliente, nombre_cliente, direccion_cliente, telefono_cliente',
    [cliente.nombreCliente, cliente.direccionCliente, cliente.telefonoCliente]
  );
  if (!row) {
    throw new Error('Error al crear el cliente');
  }
  return mapRowToCliente(row);
};

export const update = async (
  codigo: number,
  cliente: Partial<CreateClienteDTO>
): Promise<Cliente | null> => {
  const fields: string[] = [];
  const values: unknown[] = [];
  let paramIndex = 1;

  if (cliente.nombreCliente !== undefined) {
    fields.push(`nombre_cliente = $${paramIndex++}`);
    values.push(cliente.nombreCliente);
  }
  if (cliente.direccionCliente !== undefined) {
    fields.push(`direccion_cliente = $${paramIndex++}`);
    values.push(cliente.direccionCliente);
  }
  if (cliente.telefonoCliente !== undefined) {
    fields.push(`telefono_cliente = $${paramIndex++}`);
    values.push(cliente.telefonoCliente);
  }

  if (fields.length === 0) {
    return findByCodigo(codigo);
  }

  values.push(codigo);
  const row = await queryOne<ClienteRow>(
    `UPDATE clientes SET ${fields.join(', ')} WHERE codigo_cliente = $${paramIndex} RETURNING codigo_cliente, nombre_cliente, direccion_cliente, telefono_cliente`,
    values
  );
  return row ? mapRowToCliente(row) : null;
};

export const remove = async (codigo: number): Promise<boolean> => {
  const rowCount = await execute(
    'DELETE FROM clientes WHERE codigo_cliente = $1',
    [codigo]
  );
  return rowCount > 0;
};

export const existsByCodigo = async (codigo: number): Promise<boolean> => {
  const row = await queryOne<{ count: number }>(
    'SELECT COUNT(*) as count FROM clientes WHERE codigo_cliente = $1',
    [codigo]
  );
  return row ? parseInt(String(row.count), 10) > 0 : false;
};
```

```typescript
import { Pool } from 'pg';
import * as fs from 'fs';
import * as path from 'path';

const pool = new Pool({
  host: process.env.DB_HOST || '127.0.0.1',
  port: parseInt(process.env.DB_PORT || '5432', 10),
  user: process.env.DB_USER || 'postgres',
  password: process.env.DB_PASSWORD || 'admin',
  database: process.env.DB_NAME || 'clientes_db',

const createTableQuery = `
  CREATE TABLE IF NOT EXISTS clientes (
    codigo_cliente SERIAL PRIMARY KEY,
    nombre_cliente VARCHAR(100) NOT NULL,
    direccion_cliente VARCHAR(200) NOT NULL,
    telefono_cliente VARCHAR(20) NOT NULL
  );
`;

const seedData = async (): Promise<void> => {
  const seedPath = path.join(__dirname, '../../data/clientes.json');
  if (!fs.existsSync(seedPath)) {
    console.log('No seed data file found');
    return;
  }

  const rawData = fs.readFileSync(seedPath, 'utf-8');
  const clientes = JSON.parse(rawData) as Array<{
    codigo_cliente: number;
    nombre_cliente: string;
    direccion_cliente: string;
    telefono_cliente: string;
  }>;

  const client = await pool.connect();
  try {
    for (const cliente of clientes) {
      await client.query(
        `INSERT INTO clientes (codigo_cliente, nombre_cliente, direccion_cliente, telefono_cliente)
         VALUES ($1, $2, $3, $4)
         ON CONFLICT (codigo_cliente) DO NOTHING`,
        [
          cliente.codigo_cliente,
          cliente.nombre_cliente,
          cliente.direccion_cliente,
          cliente.telefono_cliente,
        ]
      );
    }
    console.log(`Seeded ${clientes.length} clients`);
  } finally {
    client.release();
  }
};

const initDatabase = async (): Promise<void> => {
  try {
    console.log('Initializing database...');
    await pool.query(createTableQuery);
    console.log('Table created successfully');

    console.log('Seeding data...');
    await seedData();

    console.log('Resetting sequence...');
    await pool.query(
      `SELECT setval('clientes_codigo_cliente_seq', COALESCE((SELECT MAX(codigo_cliente) FROM clientes), 1))`
    );
    console.log('Sequence reset successfully');

    console.log('Database initialization completed');
  } catch (error) {
    console.error('Error initializing database:', error);
    throw error;
  } finally {
    await pool.end();
  }
};
```

```typescript
import { Request, Response } from 'express';
import * as clienteRepository from '../persistence/cliente.repository';

type ValidationErrors = string[];

type RouteParams = { codigo: string };

const parseCodigo = (raw: string | string[]): number => {
  const value = Array.isArray(raw) ? raw[0] : raw;
  return parseInt(value, 10);
};

const validateCliente = (data: ClienteDTO, isUpdate = false): ValidationErrors => {
  const errors: ValidationErrors = [];

  if (!isUpdate || data.nombreCliente !== undefined) {
    if (!data.nombreCliente || data.nombreCliente.trim().length === 0) {
      errors.push('El nombre del cliente es requerido');
    } else if (data.nombreCliente.length > 100) {
      errors.push('El nombre del cliente no puede exceder 100 caracteres');
    }
  }

  if (!isUpdate || data.direccionCliente !== undefined) {
    if (!data.direccionCliente || data.direccionCliente.trim().length === 0) {
      errors.push('La dirección del cliente es requerida');
    } else if (data.direccionCliente.length > 200) {
      errors.push('La dirección del cliente no puede exceder 200 caracteres');
    }
  }

  if (!isUpdate || data.telefonoCliente !== undefined) {
    if (!data.telefonoCliente || data.telefonoCliente.trim().length === 0) {
      errors.push('El teléfono del cliente es requerido');
    } else if (data.telefonoCliente.length < 7 || data.telefonoCliente.length > 20) {
      errors.push('El teléfono debe tener entre 7 y 20 caracteres');
    } else if (!/^[0-9+\-\s()]+$/.test(data.telefonoCliente)) {
      errors.push('El teléfono solo puede contener números, espacios, guiones, paréntesis y el signo +');
    }
  }

  return errors;
};

export const createCliente = async (req: Request, res: Response): Promise<void> => {

        success: false,
        error: validationErrors.join(', '),

      direccionCliente: direccionCliente.trim(),

    const response: ApiResponse<Cliente> = {
      success: true,
      data: cliente,
      message: 'Cliente creado exitosamente',
    };
    res.status(201).json(response);

    const message = error instanceof Error ? error.message : 'Error al crear el cliente';
    const response: ApiResponse<never> = {
      success: false,
      error: message,
    };
    res.status(500).json(response);

    const codigo = parseCodigo(req.params.codigo);

      return;

        success: false,
        error: `No se encontró el cliente con código ${codigo}`,
      };
      res.status(404).json(response);
      return;
    }

    const { nombreCliente, direccionCliente, telefonoCliente } = req.body;
    const clienteData: ClienteDTO = { nombreCliente, direccionCliente, telefonoCliente };

    const validationErrors = validateCliente(clienteData, true);

      const response: ApiResponse<never> = {
        success: false,
        error: validationErrors.join(', '),
      };

      return;

    if (direccionCliente !== undefined) {

      updateData.telefonoCliente = telefonoCliente.trim();

      const response: ApiResponse<never> = {
        success: false,
        error: 'Error al actualizar el cliente',

      return;
    }

    const response: ApiResponse<Cliente> = {
      success: true,
      data: cliente,
      message: 'Cliente actualizado exitosamente',

      error: message,
```

```typescript
import { Injectable, signal } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable, tap } from 'rxjs';
import { Cliente, ClienteDTO, ApiResponse } from '../models/cliente.model';

@Injectable({
  providedIn: 'root',
})
export class ClienteService {
  private readonly apiUrl = 'http://localhost:3000/clientes';

  readonly loading = signal<boolean>(false);
  readonly error = signal<string | null>(null);
  readonly success = signal<string | null>(null);
  readonly clientesActualizados = signal<number>(0);

  constructor(private http: HttpClient) {}

  getAll(): Observable<ApiResponse<Cliente[]>> {
    this.loading.set(true);
    this.error.set(null);
    return this.http.get<ApiResponse<Cliente[]>>(this.apiUrl).pipe(
      tap({
        next: () => this.loading.set(false),
        error: () => this.loading.set(false),
      })
    );
  }

  getByCodigo(codigo: number): Observable<ApiResponse<Cliente>> {
    this.loading.set(true);
    this.error.set(null);
    return this.http.get<ApiResponse<Cliente>>(`${this.apiUrl}/${codigo}`).pipe(
      tap({
        next: () => this.loading.set(false),
        error: () => this.loading.set(false),
      })
    );
  }

  create(cliente: ClienteDTO): Observable<ApiResponse<Cliente>> {
    this.loading.set(true);
    this.error.set(null);
    this.success.set(null);
    return this.http.post<ApiResponse<Cliente>>(this.apiUrl, cliente).pipe(
      tap({
        next: (response) => {
          this.loading.set(false);
          if (response.success) {
            this.clientesActualizados.update(v => v + 1);
            if (response.message) {
              this.success.set(response.message);
            }
          }
        },
        error: () => this.loading.set(false),
      })
    );
  }

  update(codigo: number, cliente: ClienteDTO): Observable<ApiResponse<Cliente>> {
    this.loading.set(true);
    this.error.set(null);
    this.success.set(null);
    return this.http.put<ApiResponse<Cliente>>(`${this.apiUrl}/${codigo}`, cliente).pipe(
      tap({
        next: (response) => {
          this.loading.set(false);
          if (response.success) {
            this.clientesActualizados.update(v => v + 1);
            if (response.message) {
              this.success.set(response.message);
            }
          }
        },
        error: () => this.loading.set(false),
      })
    );
  }

  delete(codigo: number): Observable<ApiResponse<null>> {
    this.loading.set(true);
    this.error.set(null);
    this.success.set(null);
    return this.http.delete<ApiResponse<null>>(`${this.apiUrl}/${codigo}`).pipe(
      tap({
        next: (response) => {
          this.loading.set(false);
          if (response.success) {
            this.clientesActualizados.update(v => v + 1);
            if (response.message) {
              this.success.set(response.message);
            }
          }
        },
        error: () => this.loading.set(false),
      })
    );
  }

  clearMessages(): void {
    this.error.set(null);
    this.success.set(null);
  }

  setError(message: string): void {
    this.error.set(message);
    this.loading.set(false);
  }
}
```
